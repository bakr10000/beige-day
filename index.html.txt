        function updateClock() {

        /* ==========================================================================
           DAILY TASKS & TO-DO MANAGER
           ========================================================================== */
        function addNewTask(event) {
            event.preventDefault();
            const input = document.getElementById('newTaskInput');
            const prioritySelect = document.getElementById('newTaskPriority');
            const categorySelect = document.getElementById('newTaskCategory');
            
            const title = input.value.trim();
            if (!title) return;

            const newTask = {
                id: Date.now(),
                title: title,
                priority: prioritySelect.value,
                category: categorySelect.value,
                completed: false
            };

            state.tasks.unshift(newTask);
            saveTasks();
            renderTasks();
            input.value = '';
        }

        function toggleTask(id) {
            state.tasks = state.tasks.map(t => t.id === id ? { ...t, completed: !t.completed } : t);
            saveTasks();
            renderTasks();
        }

        function deleteTask(id) {
            state.tasks = state.tasks.filter(t => t.id !== id);
            saveTasks();
            renderTasks();
        }

        function clearCompletedTasks() {
            state.tasks = state.tasks.filter(t => !t.completed);
            saveTasks();
            renderTasks();
        }

        function setTaskFilter(filter) {
            currentTaskFilter = filter;
            
            ['All', 'Active', 'Completed'].forEach(f => {
                const btn = document.getElementById('filter' + f);
                if (btn) {
                    if (f.toLowerCase() === filter) {
                        btn.className = 'px-3 py-1 rounded-xl transition font-medium bg-cream dark:bg-beige-800 shadow-sm text-walnut dark:text-beige-100';
                    } else {
                        btn.className = 'px-3 py-1 rounded-xl transition font-medium text-beige-600 hover:text-walnut';
                    }
                }
            });

            renderTasks();
        }

        function renderTasks() {
            const container = document.getElementById('tasksList');
            container.innerHTML = '';

            let filtered = state.tasks;
            if (currentTaskFilter === 'active') {
                filtered = state.tasks.filter(t => !t.completed);
            } else if (currentTaskFilter === 'completed') {
                filtered = state.tasks.filter(t => t.completed);
            }

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-8 text-beige-400 text-xs">
                        <i class="fa-regular fa-clipboard text-2xl mb-2 block"></i>
                        <span>No tasks found</span>
                    </div>
                `;
            } else {
                filtered.forEach(task => {
                    let priorityBadge = '';
                    if (task.priority === 'high') priorityBadge = '<span class="w-2 h-2 rounded-full bg-red-400 inline-block" title="High Priority"></span>';
                    else if (task.priority === 'medium') priorityBadge = '<span class="w-2 h-2 rounded-full bg-amber-400 inline-block" title="Medium Priority"></span>';
                    else priorityBadge = '<span class="w-2 h-2 rounded-full bg-emerald-400 inline-block" title="Low Priority"></span>';

                    const item = document.createElement('div');
                    item.className = `flex items-center justify-between p-3.5 rounded-2xl bg-cream dark:bg-beige-900 border border-beige-200/80 dark:border-beige-800/80 transition ${task.completed ? 'opacity-60 line-through' : ''}`;
                    item.innerHTML = `
                        <div class="flex items-center gap-3 flex-grow pr-2">
                            <input type="checkbox" ${task.completed ? 'checked' : ''} onchange="toggleTask(${task.id})" class="custom-checkbox shrink-0">
                            <div class="flex items-center gap-2 flex-wrap">
                                <span class="text-sm font-medium text-walnut dark:text-beige-100 break-all">${escapeHtml(task.title)}</span>
                                <span class="text-[10px] px-2 py-0.5 rounded-full bg-beige-200/60 dark:bg-beige-800 text-beige-600 dark:text-beige-300 font-medium">${task.category || 'Personal'}</span>
                            </div>
                        </div>
                        <div class="flex items-center gap-2 shrink-0">
                            ${priorityBadge}
                            <button onclick="deleteTask(${task.id})" class="no-print p-1 text-beige-400 hover:text-terracotta transition text-xs">
                                <i class="fa-regular fa-trash-can"></i>
                            </button>
                        </div>
                    `;
                    container.appendChild(item);
                });
            }

            // Update Progress Bar & Stats
            const completedCount = state.tasks.filter(t => t.completed).length;
            const totalCount = state.tasks.length;
            const percentage = totalCount === 0 ? 0 : Math.round((completedCount / totalCount) * 100);

            document.getElementById('taskProgressPercent').innerText = `${percentage}%`;
            document.getElementById('taskProgressBar').style.width = `${percentage}%`;
            document.getElementById('taskStatsText').innerText = `${completedCount} of ${totalCount} completed`;
        }

        function saveTasks() {
            localStorage.setItem('beige_tasks', JSON.stringify(state.tasks));
        }

        /* ==========================================================================

        function renderHabits() {

        function toggleHabit(id) {
            state.habits = state.habits.map(h => {
                if (h.id === id) {
                    const newStatus = !h.completedToday;
                    return {
                        ...h,
                        completedToday: newStatus,
                        streak: newStatus ? h.streak + 1 : Math.max(0, h.streak - 1)
                    };
                }
                return h;
            });
            saveHabits();
            renderHabits();
        }

        function addNewHabitPrompt() {
            const habitName = prompt('Enter new habit name:');
            if (habitName && habitName.trim()) {
                state.habits.push({
                    id: Date.now(),
                    name: habitName.trim(),
                    streak: 0,
                    completedToday: false

                renderHabits();

        }

        }

        /* ==========================================================================
           POMODORO TIMER FEATURE
           ========================================================================== */
        function setPomodoroMode(mode) {
            pomoCurrentMode = mode;
            pomoTimeLeft = pomoModes[mode];
            
            if (pomoIsRunning) {
                togglePomodoro();
            }

            ['Work', 'Short', 'Long'].forEach(m => {
                const btnKey = 'pomoMode' + m;
                const btn = document.getElementById(btnKey);
                if (btn) {
                    if (m.toLowerCase() === mode.toLowerCase() || (m === 'Short' && mode === 'shortBreak') || (m === 'Long' && mode === 'longBreak')) {
                        btn.className = 'px-3.5 py-1.5 rounded-xl transition bg-cream dark:bg-beige-800 text-walnut dark:text-beige-100 shadow-sm';
                    } else {
                        btn.className = 'px-3.5 py-1.5 rounded-xl transition text-beige-600 hover:text-walnut';
                    }
                }
            });

            updatePomodoroDisplay();
        }

        function togglePomodoro() {
            if (pomoIsRunning) {
                clearInterval(pomoTimerInterval);
                pomoIsRunning = false;
                document.getElementById('pomoStartText').innerText = 'Start Focus';
                document.getElementById('pomoStartIcon').className = 'fa-solid fa-play text-xs';
                document.getElementById('pomoStateLabel').innerText = 'Paused';
            } else {
                pomoIsRunning = true;
                document.getElementById('pomoStartText').innerText = 'Pause';
                document.getElementById('pomoStartIcon').className = 'fa-solid fa-pause text-xs';
                document.getElementById('pomoStateLabel').innerText = 'Focus Mode Active';

                pomoTimerInterval = setInterval(() => {
                    if (pomoTimeLeft > 0) {
                        pomoTimeLeft--;
                        updatePomodoroDisplay();
                    } else {
                        clearInterval(pomoTimerInterval);
                        pomoIsRunning = false;
                        playChimeSound();
                        alert('Great job! Focus session completed.');
                        
                        if (pomoCurrentMode === 'work') {
                            state.pomoSessions++;
                            localStorage.setItem('beige_pomo_sessions', state.pomoSessions);
                        }
                        
                        resetPomodoro();
                    }
                }, 1000);
            }
        }

        function resetPomodoro() {
            if (pomoIsRunning) {
                clearInterval(pomoTimerInterval);
                pomoIsRunning = false;
            }
            pomoTimeLeft = pomoModes[pomoCurrentMode];
            document.getElementById('pomoStartText').innerText = 'Start Focus';
            document.getElementById('pomoStartIcon').className = 'fa-solid fa-play text-xs';
            document.getElementById('pomoStateLabel').innerText = 'Ready to Start';
            updatePomodoroDisplay();
        }

        function updatePomodoroDisplay() {
            const minutes = Math.floor(pomoTimeLeft / 60);
            const seconds = pomoTimeLeft % 60;
            const formattedTime = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
            
            document.getElementById('pomoTimeDisplay').innerText = formattedTime;

            const totalDuration = pomoModes[pomoCurrentMode];
            const progressRatio = (totalDuration - pomoTimeLeft) / totalDuration;
            const circle = document.getElementById('pomoProgressRing');
            const circumference = 2 * Math.PI * 42;
            const strokeDashoffset = circumference - (progressRatio * circumference);
            circle.style.strokeDashoffset = strokeDashoffset;

            document.getElementById('pomodoroSessionCount').innerText = `${state.pomoSessions} sessions`;
        }

        function playChimeSound() {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(587.33, ctx.currentTime);
                gain.gain.setValueAtTime(0.3, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 1.5);
                osc.connect(gain);
                gain.connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + 1.5);
            } catch(e) {
                console.log('Web Audio API notification audio not supported');
            }
        }

        function resetDayData() {

                renderTasks();
            }

        function exportData() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(state, null, 2));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", `BEIGE_DAY_Backup_${new Date().toISOString().slice(0,10)}.json`);
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        function importData(event) {
            const fileReader = new FileReader();
            fileReader.onload = function(e) {
                try {
                    const importedState = JSON.parse(e.target.result);
                    if (importedState && typeof importedState === 'object') {
                        state = { ...state, ...importedState };
                        
                        localStorage.setItem('beige_tasks', JSON.stringify(state.tasks));
                        localStorage.setItem('beige_priorities', JSON.stringify(state.topPriorities));
                        localStorage.setItem('beige_habits', JSON.stringify(state.habits));
                        localStorage.setItem('beige_gratitude', JSON.stringify(state.gratitudes));
                        localStorage.setItem('beige_notes', state.dailyNotes);
                        localStorage.setItem('beige_pomo_sessions', state.pomoSessions);

                        location.reload();
                    }
                } catch (err) {
                    alert('Invalid JSON backup file!');
                }
            };
            fileReader.readAsText(event.target.files[0]);
        }

        function escapeHtml(str) {
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
